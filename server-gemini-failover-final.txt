function validPlayerId(value) {
  return PLAYER_ID_RE.test(String(value || ""));
}

      desc: "恢复 30 点生命值"
    },

      desc: "来自农场的纪念品"
    },

function defaultGameState(player) {
  return {
    page: "home",
    level: 1,
    exp: 20,
    love: Number(player.love || 20),
    relationship: player.relationship || "初次相遇",
    hp: 100,
    maxHp: 100,
    gold: 120,
    farmCount: 0,
    monsterHp: 60,
    monsterMaxHp: 60,
    inventory: defaultInventory(),
    chatHistory: [
      {
        role: "partner",
        text: "你回来啦。今天也想和我一起冒险吗？ ❤️"
      }
    ]
  };
}

function clampInt(value, min, max, fallback) {
  const n = Number(value);
  if (!Number.isFinite(n)) return fallback;
  return Math.max(min, Math.min(max, Math.floor(n)));
}

function sanitizeInventory(value, fallback = defaultInventory()) {
  const base = new Map(defaultInventory().map((item) => [item.id, item]));
  const source = Array.isArray(value) ? value : fallback;

  return [...base.values()].map((item) => {
    const incoming = source.find((x) => x?.id === item.id);
    return {
      ...item,
      count: clampInt(incoming?.count, 0, 999999, item.count)
    };
  });
}

function sanitizeState(input, player, fallback = defaultGameState(player)) {
  const src = input && typeof input === "object" ? input : {};
  const level = clampInt(src.level, 1, 999, fallback.level);
  const maxHp = clampInt(
    src.maxHp,
    100,
    99999,
    100 + (level - 1) * 10
  );
  const allowedPages = new Set([
    "home",
    "farm",
    "map",
    "chat",
    "battle",
    "bag"
  ]);

  return {
    page: allowedPages.has(src.page) ? src.page : fallback.page,
    level,
    exp: clampInt(src.exp, 0, 9999999, fallback.exp),
    love: Number(player.love || fallback.love),
    relationship: player.relationship || fallback.relationship,
    hp: clampInt(src.hp, 0, maxHp, fallback.hp),
    maxHp,
    gold: clampInt(src.gold, 0, 999999999, fallback.gold),
    farmCount: clampInt(
      src.farmCount,
      0,
      999999999,
      fallback.farmCount
    ),
    monsterHp: clampInt(src.monsterHp, 0, 99999, fallback.monsterHp),
    monsterMaxHp: clampInt(
      src.monsterMaxHp,
      1,
      99999,
      fallback.monsterMaxHp
    ),
    inventory: sanitizeInventory(src.inventory, fallback.inventory),
    chatHistory: Array.isArray(src.chatHistory)
      ? src.chatHistory
          .slice(-50)
          .map((x) => ({
            role: x?.role === "player" ? "player" : "partner",
            text: String(x?.text || "").slice(0, 1000)
          }))
          .filter((x) => x.text)
      : fallback.chatHistory
  };
}

function loadState(id) {
  const player = ensurePlayer(id);
  const stored = getGameState(id);
  const state = sanitizeState(
    stored || {},
    player,
    defaultGameState(player)
  );

  if (!stored) {
    setGameState(id, state);
  }

  return { player, state };
}

function addItem(state, id, amount = 1) {
  const item = state.inventory.find((x) => x.id === id);
  if (item) {
    item.count = Math.max(0, item.count + amount);
  }
}

function addExp(state, amount) {
  state.exp += amount;

  while (true) {
    const needed = 100 + (state.level - 1) * 40;
    if (state.exp < needed) break;

    state.exp -= needed;
    state.level += 1;
    state.maxHp += 10;
    state.hp = state.maxHp;
  }
}

async function callGemini(instructions, message) {
  let lastError = null;

  for (const model of GEMINI_MODELS) {
    const controller = new AbortController();
    const timeout = setTimeout(() => controller.abort(), 25000);

    try {
      const endpoint =
        `https://generativelanguage.googleapis.com/v1beta/models/` +
        `${encodeURIComponent(model)}:generateContent`;

      const response = await fetch(endpoint, {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          "x-goog-api-key": process.env.GEMINI_API_KEY
        },
        signal: controller.signal,
        body: JSON.stringify({
          systemInstruction: {
            parts: [{ text: instructions }]
          },
          contents: [
            {
              role: "user",
              parts: [{ text: message }]
            }
          ],
          generationConfig: {
            temperature: 0.85,
            maxOutputTokens: 700
          }
        })
      });

      const data = await response.json().catch(() => ({}));

      if (!response.ok) {
        const detail =
          data?.error?.message ||
          data?.message ||
          `Gemini HTTP ${response.status}`;

        const error = new Error(detail);
        error.status = response.status;
        error.provider = "gemini";
        error.model = model;
        throw error;
      }

      const reply = data?.candidates?.[0]?.content?.parts
        ?.map((part) => part?.text || "")
        .join("")
        .trim();

      if (!reply) {
        const error = new Error("Gemini returned an empty response");
        error.provider = "gemini";
        error.model = model;
        throw error;
      }

      console.log(`Gemini success: ${model}`);
      return { reply, model };
    } catch (error) {
      const normalizedError =
        error?.name === "AbortError"
          ? Object.assign(new Error(`Gemini ${model} timed out`), {
              status: 504,
              provider: "gemini",
              model
            })
          : error;

      lastError = normalizedError;

      console.error("gemini_model_failed", {
        model,
        message: normalizedError?.message,
        status: normalizedError?.status || null
      });

      // Retry/fail over only for model availability, rate-limit,
      // overload, and transient upstream failures.
      if ([404, 429, 500, 502, 503, 504].includes(normalizedError?.status)) {
        continue;
      }

      // Authentication/permission and request errors should surface immediately.
      throw normalizedError;
    } finally {
      clearTimeout(timeout);
    }
  }

  throw lastError || new Error("All Gemini models failed");
}

app.get("/api/game/state", (req, res) => {
  const id = String(req.query.playerId || "");

  if (!validPlayerId(id)) {
    return res.status(400).json({ error: "invalid_player_id" });
  }

  const { state } = loadState(id);
  res.json({ state });
});

app.put("/api/game/state", (req, res) => {
  const id = String(req.body?.playerId || "");

  if (!validPlayerId(id)) {
    return res.status(400).json({ error: "invalid_player_id" });
  }

  const { player, state: current } = loadState(id);
  const state = sanitizeState(req.body?.state, player, current);

  setGameState(id, state);

  res.json({
    state,
    message: "☁️ 游戏进度已保存到服务器。"
  });
});

app.delete("/api/game/state", (req, res) => {
  const id = String(req.query.playerId || "");

  if (!validPlayerId(id)) {
    return res.status(400).json({ error: "invalid_player_id" });
  }

  deleteGameState(id);
  res.json({ ok: true });
});

app.post("/api/game/action", (req, res) => {
  try {
    const id = String(req.body?.playerId || "");
    const action = String(req.body?.action || "");

    if (!validPlayerId(id)) {
      return res.status(400).json({ error: "invalid_player_id" });
    }

    const { state } = loadState(id);
    let message = "";

    if (action === "harvest") {
      state.farmCount += 1;
      state.gold += 8;
      addExp(state, 8);
      addItem(state, "flower", 1);

      const lines = [
        "🌸 你在农场发现了一朵漂亮的爱心花。",
        "🥕 今天的作物长势很好，收获满满。",
        "🌿 风吹过农田，远处传来璃月的笑声。"
      ];

      message = lines[(state.farmCount - 1) % lines.length];
    } else if (action === "resetMonster") {
      state.monsterMaxHp =
        60 + Math.max(0, state.level - 1) * 10;
      state.monsterHp = state.monsterMaxHp;
      message = "🌙 一只新的月影史莱姆出现了！";
    } else if (action === "attack") {
      if (state.hp <= 0) {
        message = "💤 你已经没有体力了，先使用药水恢复吧。";
      } else if (state.monsterHp <= 0) {
        message = "✅ 这只魔物已经被击败，点击“寻找新敌人”。";
      } else {
        const playerDamage =
          10 + Math.floor(Math.random() * 12) + state.level * 2;

        state.monsterHp = Math.max(
          0,
          state.monsterHp - playerDamage
        );

        if (state.monsterHp <= 0) {
          const reward = 22 + state.level * 4;
          state.gold += reward;
          addExp(state, 25);
          addItem(state, "crystal", 1);
          message =
            `🏆 战斗胜利！获得 ${reward} 金币和 1 枚月光水晶。`;
        } else {
          const enemyDamage =
            5 + Math.floor(Math.random() * 9);

          state.hp = Math.max(0, state.hp - enemyDamage);

          message =
            state.hp <= 0
              ? `💥 你造成 ${playerDamage} 伤害，但被魔物击倒了。`
              : `⚔️ 你造成 ${playerDamage} 伤害，受到 ${enemyDamage} 伤害。`;
        }
      }
    } else if (action === "usePotion") {
      const potion = state.inventory.find(
        (x) => x.id === "potion"
      );

      if (!potion || potion.count <= 0) {
        message = "🧪 治疗药水已经用完了。";
      } else if (state.hp >= state.maxHp) {
        message = "❤️ 生命值已经是满的。";
      } else {
        potion.count -= 1;

        const before = state.hp;
        state.hp = Math.min(
          state.maxHp,
          state.hp + 30
        );

        message =
          `🧪 使用治疗药水，恢复了 ${state.hp - before} 点生命值。`;
      }
    } else {
      return res.status(400).json({
        error: "unknown_action"
      });
    }

    setGameState(id, state);
    res.json({ state, message });
  } catch (error) {
    console.error("game_action_failed", error);
    res.status(500).json({
      error: "game_action_failed"
    });
  }
});

app.post("/api/chat", async (req, res) => {
  try {
    const id = String(req.body?.playerId || "");
    const msg = String(req.body?.message || "").trim();

    if (
      !validPlayerId(id) ||
      !msg ||
      msg.length > 1000
    ) {
      return res.status(400).json({
        error: "invalid_request"
      });
    }

    const player = ensurePlayer(id);
    const history = recentMessages(id);
    const memory = memories(id);
    const gameState = getGameState(id);

    const gameContext = gameState
      ? `\n当前游戏状态: Lv.${gameState.level}，生命 ${gameState.hp}/${gameState.maxHp}，金币 ${gameState.gold}，农场收获 ${gameState.farmCount} 次。`
      : "";

    const instructions =
      `你是《恋爱大陆 Online》的虚构角色${character.name}。` +
      `${character.personality}\n` +
      `当前关系:${player.relationship}\n` +
      `好感:${player.love}/1000${gameContext}\n` +
      `长期记忆:\n` +
      `${memory.map((x) => "- " + x.content).join("\n") || "- 暂无"}\n` +
      `最近对话:\n` +
      `${history.map((x) => x.role + ":" + x.content).join("\n") || "暂无"}\n` +
      `用自然中文回复，通常1到4段。` +
      `可以自然提到玩家刚才的冒险状态，但不要编造未发生的事件。`;

    addMessage(id, "user", msg);

    const aiResult = await callGemini(
      instructions,
      msg
    );
    const reply = aiResult.reply;

    addMessage(id, "assistant", reply);

    const love = Math.min(
      1000,
      Number(player.love || 20) + 2
    );

    const relationship = relationshipFor(love);
    updateRelationship(
      id,
      love,
      relationship
    );

    if (
      /喜欢|最爱|生日|名字|讨厌|害怕|梦想|工作|家乡/.test(
        msg
      )
    ) {
      addMemory(id, msg, 3);
    }

    const latest = getGameState(id);

    if (latest) {
      latest.love = love;
      latest.relationship = relationship;
      setGameState(id, latest);
    }

    res.json({
      reply,
      love,
      relationship,
      model: aiResult.model
    });
  } catch (error) {
    console.error("chat_failed", {
      message: error?.message,
      status: error?.status,
      provider: error?.provider || "gemini"
    });

    res.status(500).json({
      error: "chat_failed"
    });
  }
});
